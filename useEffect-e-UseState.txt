// Importa os Hooks useState e useEffect da biblioteca React
import { useEffect, useState } from 'react'
// Importa a imagem do logo React (não está sendo usada no código)
import reactLogo from './assets/react.svg'
// Importa a imagem do logo Vite (não está sendo usada no código)
import viteLogo from '/vite.svg'
// Importa o arquivo de estilos CSS
import './App.css'

// Declaração do componente principal App
function App() {
  // Declara uma variável de estado 'count' com valor inicial 0
  // setCount é a função que atualiza o valor de count
  const [count, setCount] = useState(0)

  // useEffect que executa uma função sempre que o 'count' mudar
  // O array [count] nas dependências faz ele monitorar mudanças no count
  useEffect(() => {
    // Exibe uma mensagem no console com o valor atual do contador
    console.log('Contador atualizado:', count)
  }, [count]) // Dependência: executa quando count alterar

  // JSX: o que será renderizado na tela
  return (
    <div>
      {/* Título da aplicação */}
      <h1>Sistema de mercado</h1>

      {/* Exibe o valor atual do contador */}
      <h2>Contador: {count}</h2>

      {/* Botão que ao ser clicado chama setCount aumentando o valor em 1 */}
      <button onClick={() => setCount(count + 1)}>
        Aumentar
      </button>
    </div>
  )
}

// Exporta o componente para ser usado em outros arquivos
export default App
